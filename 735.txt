class Solution {
    public int[] asteroidCollision(int[] asteroids) {
        Stack<Integer> st = new Stack<>();
        int left = 0;
        while(left<asteroids.length){
            if(st.isEmpty()){
                st.push(asteroids[left]);
            }
            else if(st.peek()<0){
                st.push(asteroids[left]);
            }
            else if(asteroids[left]+st.peek()==0){
                st.pop();
            }
            else if(asteroids[left]>0 && st.peek()>0){
                st.push(asteroids[left]);
            }
            else if(asteroids[left]<0 && st.peek()>0){
                if(asteroids[left]+st.peek()<0){
                    st.pop();
                    left--;
                }
            }
            else if(asteroids[left]>0 && st.peek()<0){
                st.push(asteroids[left]);
            }
            left++;
        }
        int[] res = new int[st.size()];
        for(int i=res.length-1;i>=0;i--){
            res[i] = st.pop();
        }
        return res;
    }
}
