class Solution {
    public int compress(char[] chars) {
        int left = 0;
        int count = 0;
        int i = 0;
        while(left<chars.length){
            count = 1;
            while(left<chars.length-1 && chars[left+1]==chars[left]){
                count++;
                left++;
            }
            chars[i] = chars[left];    i++;
            if(count>1){
                String s = new String(String.valueOf(count));
                for(char c:s.toCharArray()){
                    chars[i] = c;  i++;
                }
            } 
            left++;
        }
        
        return i;
    }
}
