import java.util.*;

public class Main {
    public static void main(String[] args) {
      int num = 356;
      for(int i=1;i<=5;i++){
        int org = num;
        int rev = 0;
        while(num!=0){
          int last = num%10;
          rev = rev*10+last;
          num = num/10;
        }
        int sum = rev+org;
        int revsum = 0;
        int dummy = sum;
        while(dummy!=0){
          int last = dummy%10;
          revsum = revsum*10+last;
          dummy = dummy/10;
        }
        System.out.println(sum+"+"+revsum);
        if(sum==revsum){
          System.out.println("It is a palindrome");
          break;
        }
        else num=sum;
      }
    
    }
}
