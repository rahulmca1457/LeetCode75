class Solution {
    public boolean isValid(String s) {
        Stack<Character> st = new Stack<>();
        if(s.length()==1) return false;
        for(char c:s.toCharArray()){
            if(c=='(' || c=='{' || c=='['){
                st.push(c);
            }
            else{
                if(!st.isEmpty() && c==')' && st.peek()=='('){
                    st.pop();
                    continue;
                }
                else if(!st.isEmpty() && c=='}' && st.peek()=='{'){
                    st.pop();
                    continue;
                }
                else if(!st.isEmpty() && c==']' && st.peek()=='['){
                    st.pop();
                    continue;
                }
                else return false;
            }
        }
        return st.isEmpty();
    }
}
