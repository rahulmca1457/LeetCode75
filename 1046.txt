class Solution {
    public int lastStoneWeight(int[] stones) {
        PriorityQueue<Integer> max = new PriorityQueue<>(Collections.reverseOrder());
        for(int i:stones){
            max.offer(i);
        }
        while(max.size()>1){
            int y = max.poll();
            int x = max.poll();
            if(y>x){
                y = y-x;
                max.offer(y);
            }
        }
        if(max.size()==0) return 0;
        return max.poll();
    }
}
