class Solution {
    public int pivotIndex(int[] nums) {
        int[] pre = new int[nums.length];
        int[] post = new int[nums.length];
        int result = -1;
        //prefix sum
        int sum = 0;
        for(int i=0;i<nums.length;i++){
            sum+=nums[i];
            pre[i] = sum;
        }
        sum = 0;
        for(int i=nums.length-1;i>=0;i--){
            sum+=nums[i];
            post[i] = sum;
            if(pre[i]==post[i]) result = i;
        }
        return result;
    }
}
