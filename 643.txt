class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int left = 0;
        int right = k;
        double sum = 0;
        double max = Integer.MIN_VALUE;
        for(int i=0;i<k;i++){
            sum+=nums[i];
        }
        max = Math.max(max,sum/k);
        while(right<nums.length){
            sum = sum-nums[left];
            sum = sum+nums[right];
            max = Math.max(max,sum/k);
            left++;
            right++;
        }
        return max;
    }
}
