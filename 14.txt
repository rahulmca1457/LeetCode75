class Solution {
    public String longestCommonPrefix(String[] strs) {
        String s = strs[0];
        int i = 0;
        while(i<strs.length){
            if(strs[i].indexOf(s)==0){
                i++;
            }
            else if(s.length()==0){
                return "";
            }
            else{
                s = s.substring(0,s.length()-1);
            }
        }
        return s;
    }
}
