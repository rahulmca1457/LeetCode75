class Solution {
    public int numberOfSubarrays(int[] nums, int k) {
        HashMap<Integer,Integer> map = new HashMap<>();
        int count = 0;
        int oddcount = 0;
        map.put(0,1);
        for(int num:nums){
            if(num%2==1){
                oddcount++;
            }
            if(map.containsKey(oddcount-k)){
                count+=map.get(oddcount-k);
            }
            map.put(oddcount,map.getOrDefault(oddcount,0)+1);
        }
        return count;
    }
}
