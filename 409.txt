class Solution {
    public int longestPalindrome(String s) {
        HashMap<Character,Integer> map = new HashMap<>();
        int single = 0;
        int count = 0;
        for(char c:s.toCharArray()){
            map.put(c,map.getOrDefault(c,0)+1);
        }
        for(char c:map.keySet()){
            if(map.get(c)%2==0){
                count+=map.get(c);
            }
            else if(map.get(c)>2){
                count+=map.get(c)-1;
                single = 1;
            }
            else{
                single = 1;
            }
        }
        return single+count;
    }
}
