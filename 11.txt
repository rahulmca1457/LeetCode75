class Solution {
    public int maxArea(int[] height) {
        int left = 0;
        int right = height.length-1;
        int max = 0;
        while(left<right){
            int lowheight;
            if(height[left]>height[right]) lowheight = height[right];
            else lowheight = height[left];
            max = Math.max(max,(right-left)*lowheight); 
            if(height[right]>height[left]) left++;
            else right--; 
        }
        return max;
    }
}
