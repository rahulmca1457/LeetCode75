class MyStack {
    Queue<Integer> q = new LinkedList<>();
    Queue<Integer> q2 = new LinkedList<>();
    public MyStack() {
        
    }
    
    public void push(int x) {
        q.offer(x);
    }
    
    public int pop() {
        int size = q.size();
        for(int i=0;i<size-1;i++){
            q2.offer(q.poll());
        }
        int data = q.poll();
        while(!q2.isEmpty()){
            q.offer(q2.poll());
        }
        return data;
    }
    
    public int top() {
        int size = q.size();
        for(int i=0;i<size-1;i++){
            q2.offer(q.poll());
        }
        int data = q.peek();
        while(!q2.isEmpty()){
            q.offer(q2.poll());
        }
        q.offer(q.poll());
        return data;
    }
    
    public boolean empty() {
        return q.isEmpty();
    }
}
