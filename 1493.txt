class Solution {
    public int longestSubarray(int[] nums) {
        int left = 0;
        int right = 0;
        int count = 0;
        int max = 0;
        int zero = 1;
        while(right<nums.length){
            if(nums[right]==1){
                count++;
                if(count>max) max = count;
                right++;
            }
            else if(nums[right]==0 && zero>0){
                zero--;
                count++;
                if(count>max) max = count;
                right++;
            }
            else if(nums[right]==0 && zero==0){
                if(nums[left]==0){
                    zero++;
                }
                left++;
                count--;
            }
        }
        return max-1;
    }
}
