class Solution {
    public String removeStars(String s) {
        Stack<Character> st = new Stack<>();
        for(char c:s.toCharArray()){
            if(c=='*' && !st.isEmpty()){
                st.pop();
            }
            else if(c!='*')st.push(c);
        }
        String res =  "";
        while(!st.isEmpty()){
            res+=st.pop();
        } 
        String res2 = "";
        for(int i=res.length()-1;i>=0;i--){
            res2+=String.valueOf(res.charAt(i));
        }
        return res2;
    }
}
