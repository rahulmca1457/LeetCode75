class Solution {
    public int longestConsecutive(int[] nums) {
        if(nums.length==0) return 0;
        int start = 0;
        int count = 1;
        int max = 1;
        HashSet<Integer> set = new HashSet<>();
        for(int i:nums){
            set.add(i);
        }
        for(int i:set){
            if(!set.contains(i-1)){
                start = i;
                while(set.contains(start+1)){
                    count++;
                    start++;
                }
                max = Math.max(max,count);
                count = 1;
            }
        }
        return max;
    }
}
