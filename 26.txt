class Solution {
    public int removeDuplicates(int[] nums) {
        int lastseen = nums[0];
        int j = 1;
        for(int i=1;i<nums.length;i++){
            if(nums[i]==lastseen){
                continue;
            }
            else{
                lastseen = nums[i];
                nums[j] = nums[i];
                j++;
            }
        }
        return j;
    }
}
